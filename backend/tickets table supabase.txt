Ticket Table creation schema:

-- =========================================
-- EXTENSIONS
-- =========================================
CREATE EXTENSION IF NOT EXISTS vector;
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- =========================================
-- DROP IF EXISTS (SAFE RESET)
-- =========================================
DROP TABLE IF EXISTS tickets CASCADE;

-- =========================================
-- TICKETS TABLE (CLEAN ARCHITECTURE)
-- =========================================
CREATE TABLE tickets (

    -- =====================================
    -- IDENTIFIERS
    -- =====================================
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    issue_key TEXT UNIQUE NOT NULL,
    child_key TEXT UNIQUE,
    parent_ticket_key TEXT,

    -- =====================================
    -- USER DETAILS
    -- =====================================
    name TEXT NOT NULL,
    email TEXT NOT NULL,

    -- =====================================
    -- TICKET CONTENT
    -- =====================================
    summary TEXT NOT NULL,
    description TEXT,

    -- =====================================
    -- APPLICATION CONTEXT
    -- =====================================
    app_name TEXT,
    app_code TEXT,

    component_name TEXT,
    component_code TEXT,

    -- =====================================
    -- PRIORITY CLASSIFICATION
    -- =====================================
    urgency TEXT NOT NULL,
    impact TEXT NOT NULL,

    -- =====================================
    -- STATUS FLOW
    -- =====================================
    status TEXT NOT NULL DEFAULT 'Open'
        CHECK (status IN ('Open', 'Open_no_SOP', 'Completed')),

    -- =====================================
    -- DUPLICATE MANAGEMENT
    -- =====================================
    is_duplicate BOOLEAN DEFAULT FALSE,
    duplicate_of TEXT,

    -- =====================================
    -- VECTOR SEARCH (EMBEDDINGS)
    -- =====================================
    embedding VECTOR(384),
    embedding_model TEXT DEFAULT 'all-MiniLM-L6-v2',

    -- =====================================
    -- SOP / RUNBOOK DATA
    -- =====================================
    sop_title TEXT,
    sop_confluence_id TEXT,
    sop_url TEXT,
    sop_category TEXT,
    match_type TEXT,
    paired_steps JSONB DEFAULT '[]'::jsonb,

    -- =====================================
    -- RCA (ROOT CAUSE ANALYSIS)
    -- =====================================
    rca_root_cause TEXT,
    rca_affected TEXT,
    rca_steps JSONB DEFAULT '[]'::jsonb,
    rca_confidence TEXT,
    rca_source TEXT,
    rca_matched_from TEXT,
    rca_matched_summary TEXT,

    -- =====================================
    -- AUDIT / TIMESTAMPS
    -- =====================================
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);


-- =========================================
-- INDEXES
-- =========================================

CREATE INDEX idx_ticket_issue
ON tickets(issue_key);

CREATE INDEX idx_ticket_status
ON tickets(status);

CREATE INDEX idx_ticket_app
ON tickets(app_name);

CREATE INDEX idx_ticket_component
ON tickets(component_name);

ticket_simlarity_srearch schema:

-- Drop old 4-arg function
DROP FUNCTION IF EXISTS match_tickets(
    VECTOR(384),
    INT,
    TEXT,
    TEXT
);

-- Create updated function
CREATE OR REPLACE FUNCTION match_tickets(
    query_embedding VECTOR(384),
    match_count INT,
    app_filter TEXT,
    component_filter TEXT
)

RETURNS TABLE (
    id UUID,
    issue_key TEXT,
    child_key TEXT,
    summary TEXT,
    status TEXT,
    similarity FLOAT
)

LANGUAGE sql
STABLE

AS $$

SELECT
    id,
    issue_key,
    child_key,
    summary,
    status,

    1 - (
        embedding <=> query_embedding
    ) AS similarity

FROM tickets

WHERE
    embedding IS NOT NULL
    AND status='Open'
    AND is_duplicate=FALSE
    AND child_key IS NULL

    -- only same app/component
    AND app_name = app_filter
    AND component_name = component_filter

ORDER BY embedding <=> query_embedding

LIMIT match_count;

$$;




-- Search Completed tickets


DROP FUNCTION IF EXISTS search_completed_tickets(
    VECTOR(384),
    INT,
    TEXT,
    TEXT
);

CREATE OR REPLACE FUNCTION search_completed_tickets(
    query_embedding VECTOR(384),
    match_count INT,
    app_filter TEXT DEFAULT NULL,
    component_filter TEXT DEFAULT NULL
)
RETURNS TABLE(
    issue_key TEXT,
    child_key TEXT,
    name TEXT,
    email TEXT,
    summary TEXT,
    description TEXT,
    status TEXT,
    app_name TEXT,
    component_name TEXT,
    similarity FLOAT
)
LANGUAGE sql
STABLE
AS $$

SELECT
    issue_key,
    child_key,
    name,
    email,
    summary,
    description,
    status,
    app_name,
    component_name,

    1 - (embedding <=> query_embedding) AS similarity

FROM tickets
WHERE
    embedding IS NOT NULL
    AND status = 'Completed'
    AND is_duplicate = FALSE
    AND child_key IS NULL

    -- STRICT APP FILTER (FIXED)
    AND (
        app_filter IS NULL
        OR LOWER(COALESCE(app_name, '')) = LOWER(app_filter)
    )

    -- STRICT COMPONENT FILTER (FIXED)
    AND (
        component_filter IS NULL
        OR LOWER(COALESCE(component_name, '')) = LOWER(component_filter)
    )

ORDER BY embedding <=> query_embedding
LIMIT match_count;

$$;
